// importaciones
import { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';
import qs from 'qs'

class Dao1 {

  private axios: AxiosInstance;
  private sessionCookieName: string;
  private sessionCookie: string;

  // constructor
  constructor(axios: AxiosInstance) {
    // biblioteca axios usada para realizar las solicitudes HTTP
    this.axios = axios;
    // cookie de sesión
    this.sessionCookieName = "PHPSESSID";
    this.sessionCookie = '';
  }

  // inicio de sesión
  async initSession(): Promise<any> {
    // opciones de la solicitud HTTP [get /main.php?action=init-session&type=json]
    const options: AxiosRequestConfig = {
      // [actualización de TypeScript] URL incluida en [options.url] (vea getRemoteData)
      url: 'main.php',
      method: "GET",
      // parámetros de la URL
      params: {
        action: 'init-session',
        type: 'json'
      }
    };
    // ejecución de la solicitud HTTP
    return await this.getRemoteData(options);
  }

  async authenticateUser(user: string, password: string): Promise<any> {
    // opciones de la solicitud HTTP [post /main.php?action=authenticate-user]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // cuerpo del POST
      data: qs.stringify({
        user: user,
        password: password
      }),
      // parámetros de la URL
      params: {
        action: 'authenticate-user'
      }
    };
    // ejecución de la solicitud HTTP
    return await this.getRemoteData(options);
  }

  // cálculo del impuesto
  async calculateTax(married: string, children: number, salary: number): Promise<any> {
    // opciones de la solicitud HTTP
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // cuerpo del POST
      data: qs.stringify({
        married: married,
        children: children,
        salary: salary
      }),
      // parámetros de la URL
      params: {
        action: 'calculate-tax'
      }
    };
    // ejecución de la solicitud HTTP
    const data = await this.getRemoteData(options);
    // resultado
    return data;
  }

  // lista de simulaciones
  async listSimulations(): Promise<any> {
    // opciones de la solicitud HTTP
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // parámetros de la URL
      params: {
        action: 'list-simulations'
      },
    };
    // ejecución de la solicitud HTTP
    const data = await this.getRemoteData(options);
    // resultado
    return data;
  }

  // eliminar una simulación
  async deleteSimulation(index: number): Promise<any> {
    // opciones de la solicitud HTTP
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // parámetros de la URL
      params: {
        action: 'delete-simulation',
        number: index
      },
    };
    // ejecución de la solicitud HTTP
    const data = await this.getRemoteData(options);
    // resultado
    return data;
  }

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // para la cookie de sesión
    if (!options.headers) {
      options.headers = {} as any;
    }
    (options.headers as any).Cookie = this.sessionCookie;
    // ejecución de la solicitud HTTP
    let response: AxiosResponse;
    try {
      // solicitud asíncrona
      response = await this.axios.request(options);
    } catch (error: any) {
      // el parámetro [error] es una instancia de excepción - puede adoptar diversas formas
      if (error.response) {
        // la respuesta del servidor está en [error.response]
        response = error.response;
      } else {
        // volvemos a lanzar el error
        throw error;
      }
    }
    // response es la respuesta HTTP completa del servidor (encabezados HTTP + la respuesta propiamente dicha)
    // recuperamos la cookie de sesión si existe
    const setCookie = response.headers['set-cookie'];
    if (setCookie) {
      // setCookie es un arreglo
      // buscamos la cookie de sesión en este arreglo
      let found = false;
      let i = 0;
      while (!found && i < setCookie.length) {
        // buscamos la cookie de sesión
        const results = RegExp('^(' + this.sessionCookieName + '.+?);').exec(setCookie[i]);
        if (results) {
          // almacenamos la cookie de sesión
          // eslint-disable-next-line require-atomic-updates
          this.sessionCookie = results[1];
          // la encontramos
          found = true;
        } else {
          // siguiente elemento
          i++;
        }
      }
    }
    // la respuesta del servidor está en [response.data]
    return response.data;
  }
}

async function main(): Promise<void> {
  // configuración de axios

  // instanciación de la capa [dao]

    // autenticación
    log("-----------authenticate-user");
    response = await dao.authenticateUser("admin", "admin");
    log(response);
    // cálculos del impuesto
    log("-----------calculate-tax x 3");
    response = await Promise.all([
      dao.calculateTax("yes", 2, 45000),
      dao.calculateTax("no", 2, 45000),

    log(response);
    // lista de simulaciones
    log("-----------list-of-simulations");
    response = await dao.listSimulations();

    // eliminación de una simulación
    log("-----------deleting simulation no. 1");
    response = await dao.deleteSimulation(1);
    log(response);
  } catch (error: any) {
    // registramos el error en el log
    console.log("error=", error.message);
  }
}

// clase de negocio
class Business {

  // [any]: la estructura exacta de los datos enviados por el servidor de impuestos (tasas, límites, ...)
  // no está documentada aquí; por eso nos mantenemos deliberadamente flexibles en este tipo
  taxAdminData: any;

  // constructor
  constructor(taxAdmindata: any) {
    // this.taxAdminData: datos de la administración tributaria
    this.taxAdminData = taxAdmindata;
  }

  // cálculo del impuesto
  // --------------------------------------------------------------------------
  calculateTax(married: string, children: number, salary: number): any {
    // married: sí, no
    // children: número de hijos
    // salary: salario anual
    // this.taxAdminData: datos de la administración tributaria
    //
    // cálculo del impuesto con hijos
    const result1 = this.calculateTax2(married, children, salary);
    const tax1 = result1["tax"];
    // cálculo del impuesto sin los hijos
    let result2, tax2, halfShareCeiling;
    if (children !== 0) {
      result2 = this.calculateTax2(married, 0, salary);
      tax2 = result2["tax"];
      // aplicación del tope del cociente familiar
      halfShareCeiling = this.taxAdminData.halfShareQfCeiling;
      if (children < 3) {
        // HALF_SHARE_QF_CEILING euros por los 2 primeros hijos
        tax2 = tax2 - children * halfShareCeiling;
      } else {
        // HALF_SHARE_QF_CEILING euros por los 2 primeros hijos, el doble por los siguientes
        tax2 = tax2 - 2 * halfShareCeiling - (children - 2) * 2 * halfShareCeiling;
      }
    } else {
      // sin recálculo del impuesto
      tax2 = tax1;
      result2 = result1;
    }
    // tomamos el impuesto más alto entre [tax1, tax2]
    let tax, rate, surcharge;
    if (tax1 > tax2) {
      tax = tax1;
      rate = result1["rate"];
      surcharge = result1["surcharge"];
    } else {
      surcharge = tax2 - tax1 + result2["surcharge"];
      tax = tax2;
      rate = result2["rate"];
    }
    // cálculo de un posible descuento
    const discount = this.getDiscount(married, tax);
    tax -= discount;
    // cálculo de una posible reducción de impuesto
    const reduction = this.getReduction(married, salary, children, tax);
    tax -= reduction;
    // resultado
    return {
      "tax": Math.floor(tax), "surcharge": surcharge, "discount": discount, "reduction": reduction,
      "rate": rate
    };
  }

  // --------------------------------------------------------------------------
  calculateTax2(married: string, children: number, salary: number): any {
    // married: sí, no
    // children: número de hijos
    // salary: salario anual
    // this->taxAdminData: datos de la administración tributaria
    // 
    // número de partes
    married = married.toLowerCase();
    let shares;
    if (married === "yes") {
      shares = children / 2 + 2;
    } else {
      shares = children / 2 + 1;
    }
    // 1 parte por hijo a partir del 3.º
    if (children >= 3) {
      // media parte más por cada hijo a partir del 3.º
      shares += 0.5 * (children - 2);
    }
    // renta imponible
    const taxableIncome = this.getTaxableIncome(salary);
    // recargo
    let surcharge = Math.floor(taxableIncome - 0.9 * salary);
    // por problemas de redondeo
    if (surcharge < 0) {
      surcharge = 0;
    }
    // cociente familiar
    const quotient = taxableIncome / shares;
    // cálculo del impuesto
    const limits = this.taxAdminData.limits;
    const coeffR = this.taxAdminData.coeffR;
    const coeffN = this.taxAdminData.coeffN;
    // se coloca al final del arreglo de límites para detener el siguiente bucle
    limits[limits.length - 1] = quotient;
    // búsqueda de la tasa del impuesto
    let i = 0;
    while (quotient > limits[i]) {
      i++;
    }
    // como colocamos quotient al final del arreglo de límites, el bucle anterior
    // no puede salirse del arreglo de límites
    // ahora podemos calcular el impuesto
    const tax = Math.floor(taxableIncome * coeffR[i] - shares * coeffN[i]);
    // resultado
    return { "tax": tax, "surcharge": surcharge, "rate": coeffR[i] };
  }

  // taxableIncome=annualSalary-allowance
  // la deducción tiene un mínimo y un máximo
  getTaxableIncome(salary: number): number {
    // deducción del 10 % del salario
    let allowance = 0.1 * salary;
    // esta deducción no puede superar taxAdminData.getTenPercentAllowanceMax()
    if (allowance > this.taxAdminData.tenPercentAllowanceMax) {
      allowance = this.taxAdminData.tenPercentAllowanceMax;
    }
    // la deducción no puede ser inferior a taxAdminData.getTenPercentAllowanceMin()
    if (allowance < this.taxAdminData.tenPercentAllowanceMin) {
      allowance = this.taxAdminData.tenPercentAllowanceMin;
    }
    // renta imponible
    const taxableIncome = salary - allowance;
    // resultado
    return Math.floor(taxableIncome);
  }

  // calcula un posible descuento
  getDiscount(married: string, taxes: number): number {
    // inicialmente, un descuento nulo
    let discount = 0;
    // monto máximo de impuesto para obtener el descuento
    let taxCeilingForDiscount = married === "yes" ?
      this.taxAdminData.coupleTaxCeilingForDiscount :
      this.taxAdminData.singleTaxCeilingForDiscount;
    let discountCeiling;
    if (taxes < taxCeilingForDiscount) {
      // monto máximo del descuento
      discountCeiling = married === "yes" ?
        this.taxAdminData.coupleDiscountCeiling :
        this.taxAdminData.singleDiscountCeiling;
      // descuento teórico
      discount = discountCeiling - 0.75 * taxes;
      // el descuento no puede superar el monto del impuesto
      if (discount > taxes) {
        discount = taxes;
      }
      // ningún descuento <0
      if (discount < 0) {
        discount = 0;
      }
    }
    // resultado
    return Math.ceil(discount);
  }

  // calcula una posible reducción
  getReduction(married: string, salary: number, children: number, taxes: number): number {
    // el tope de ingresos para tener derecho a la reducción del 20 %
    let incomeCeilingForReduction = married === "yes" ?
      this.taxAdminData.coupleIncomeCeilingForReduction :
      this.taxAdminData.singleIncomeCeilingForReduction;
    incomeCeilingForReduction += children * this.taxAdminData.halfShareReductionValue;
    if (children > 2) {
      incomeCeilingForReduction += (children - 2) * this.taxAdminData.halfShareReductionValue;
    }
    // renta imponible
    const taxableIncome = this.getTaxableIncome(salary);
    // reducción
    let reduction = 0;
    if (taxableIncome < incomeCeilingForReduction) {
      // reducción del 20 %
      reduction = 0.2 * taxes;
    }
    // resultado
    return Math.ceil(reduction);
  }
}

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // para la cookie de sesión
    if (!options.headers) {
      options.headers = {} as any;
    }
    (options.headers as any).Cookie = this.sessionCookie;
    // ejecución de la solicitud HTTP
    let response: AxiosResponse;
    try {
      // solicitud asíncrona
      response = await this.axios.request(options);
    } catch (error: any) {
      // el parámetro [error] es una instancia de excepción - puede adoptar diversas formas
      if (error.response) {
        // la respuesta del servidor está en [error.response]
        response = error.response;
      } else {
        // volvemos a lanzar el error
        throw error;
      }
    }
    // response es la respuesta HTTP completa del servidor (encabezados HTTP + la respuesta propiamente dicha)
    // recuperamos la cookie de sesión si existe
    const setCookie = response.headers['set-cookie'];
    if (setCookie) {
      // setCookie es un arreglo
      // buscamos la cookie de sesión en este arreglo
      let found = false;
      let i = 0;
      while (!found && i < setCookie.length) {
        // buscamos la cookie de sesión
        const results = RegExp('^(' + this.sessionCookieName + '.+?);').exec(setCookie[i]);
        if (results) {
          // almacenamos la cookie de sesión
          // eslint-disable-next-line require-atomic-updates
          this.sessionCookie = results[1];
          // la encontramos
          found = true;
        } else {
          // siguiente elemento
          i++;
        }
      }
    }
    // la respuesta del servidor está en [response.data]
    return response.data;
  }

Sin embargo, es posible configurar el servidor NestJS para que acepte estas solicitudes llamadas “entre dominios”. Aquí esto ya se ha hecho. // importaciones
import { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';
import qs from 'qs'

class Dao3 {

  private axios: AxiosInstance;

  // constructor
  constructor(axios: AxiosInstance) {
    this.axios = axios;
  }

  // inicio de sesión
  async initSession(): Promise<any> {
    // opciones de la solicitud HTTP [get /main.php?action=init-session&type=json]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // parámetros de la URL
      params: {
        action: 'init-session',
        type: 'json'
      }
    };
    // ejecución de la solicitud HTTP
    return await this.getRemoteData(options);
  }

  async authenticateUser(user: string, password: string): Promise<any> {
    // opciones de la solicitud HTTP [post /main.php?action=authenticate-user]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // cuerpo del POST
      data: qs.stringify({
        user: user,
        password: password
      }),
      // parámetros de la URL
      params: {
        action: 'authenticate-user'
      }
    };
    // ejecución de la solicitud HTTP
    return await this.getRemoteData(options);
  }

  async getAdminData(): Promise<any> {
    // opciones de la solicitud HTTP  [get /main.php?action=get-admindata]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // parámetros de la URL
      params: {
        action: 'get-admindata'
      }
    };
    // ejecución de la solicitud HTTP
    const data = await this.getRemoteData(options);
    // result
    return data;
  }

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // ejecución de la solicitud HTTP
    let response: AxiosResponse;
    try {
      // solicitud asíncrona
      response = await this.axios.request(options);
    } catch (error: any) {
      // el parámetro [error] es una instancia de excepción - puede adoptar diversas formas
      if (error.response) {
        // la respuesta del servidor está en [error.response]
        response = error.response;
      } else {
        // volvemos a lanzar el error
        throw error;
      }
    }
    // response es la respuesta HTTP completa del servidor (encabezados HTTP + la respuesta propiamente dicha)
    // la respuesta del servidor está en [response.data]
    return response.data;
  }
}

export default Dao3;

    6.5.1. Arquitectura de la solución

    Este cliente HTML funciona exactamente igual que [main3.ts], y esta es una decisión de diseño clave: solo llama al servidor (PHP) para tres acciones, siempre las mismas: init-session, authenticate-user y get-admindata. Sin embargo, el código del servidor (PHP) muestra que también es capaz de calcular el impuesto por sí mismo y de almacenar las simulaciones en su sesión (son las acciones calculate-tax, list-simulations y delete-simulation, que se encuentran en la carpeta [Controllers]), pero este cliente no las utiliza.

    La razón es simple: una vez que los datos fiscales (tasas, umbrales, etc.) se han obtenido una sola vez mediante get-admindata, todo lo necesario para calcular el impuesto ya está en el navegador, gracias a la clase [Business] . Solicitar este cálculo al servidor para cada simulación exigiría un viaje de ida y vuelta evitable por la red; asimismo, dado que cada simulación calculada se almacena de inmediato en el navegador, volver a pedir la lista al servidor (list-simulations) no aportaría ningún valor. Por lo tanto, este cliente minimiza al máximo las solicitudes al servidor, exactamente el principio ya implementado en [main3.ts]:

    • la capa [dao] ([Dao3.ts]) permanece sin cambios: sus tres métodos ya presentados arriba (initSession, authenticateUser y getAdminData) son suficientes; no se le agrega ninguno;
    • inmediatamente después de una autenticación exitosa, getAdminData se llama una vez por todas, y su resultado se usa para construir un objeto Business (new Business(taxAdminData)), que se conserva durante toda la sesión de trabajo;
    • cada cálculo de impuesto llama entonces directamente a business.calculateTax(married, children, salary), una función JavaScript ordinaria, ejecutada en el navegador, sin ninguna llamada de red;
    • las simulaciones calculadas se almacenan en un simple arreglo JavaScript, mantenido en memoria por el controlador: un push por cálculo, un splice por eliminación, exactamente como lo hace [main3.ts] con sus instrucciones simulations.push(...) y simulations.splice(1, 1).

    Una ligera diferencia con [main3.ts]: el método business.calculateTax devuelve únicamente los valores que ha calculado (tax, surcharge, discount, reduction, rate), no los parámetros ingresados por el usuario (married, children, salary). [main3.ts] no los necesita, ya que simplemente usa console.log; pero nuestra vista [List of simulations] debe mostrar estos tres parámetros en la tabla, de modo que el controlador se los agrega al propio resultado antes de almacenarlo (lo veremos en el código de [AppHtml.ts]).

    Por último, la vista [End of session] tampoco llama ya al servidor: simplemente descarta el objeto [Business] y la lista de simulaciones, y luego vuelve a mostrar la pantalla de autenticación. Recuerde que las respuestas de las tres acciones del servidor que siguen en uso tienen siempre el mismo formato, que ya vimos en [main3.ts]: un objeto JSON { action, status, response }, donde response contiene el resultado solicitado (si tuvo éxito) o un mensaje de error: una cadena o un arreglo de cadenas, según el controlador PHP correspondiente.

    6.5.1.4. La capa [views]: el archivo [Views.ts]

// archivo [Business.ts] que [main3.ts]) la que calcula el impuesto en el
// navegador, sin ninguna nueva llamada de red. La lista de simulaciones también
// se mantiene solo en el navegador (arreglo [this.simulations]),
// exactamente como [main3.ts] con su arreglo local [simulations]
// (véase `simulations.push(...)` y `simulations.splice(...)`).
// -----------------------------------------------------------------------
class AppHtml {

  private dao: Dao;
  private views: Views;
  // capa [business]: solo existe una vez recuperados los datos de impuestos
  // (después de una autenticación exitosa) - ver [initBusiness]
  private business?: Business;
  // lista de las simulaciones ya calculadas, mantenida en el navegador
  private simulations: Simulation[] = [];

  constructor() {
    // configuración de axios - idéntica a [main3.ts]
    axios.defaults.timeout = 5000;
    axios.defaults.baseURL = 'http://localhost/';
    axios.defaults.withCredentials = true;
    // capas [dao] y [views]
    this.dao = new Dao(axios);
    this.views = new Views(document.getElementById('app')!);
  }

  // punto de entrada de la aplicación
  async start(): Promise<void> {
    try {
      // inicio de sesión - obligatorio antes de cualquier otra acción (ver [main.php])
      const response = await this.dao.initSession();
      if (response.status !== 700) {
        this.views.authenticationView(this.formatError(response.response), {
          onSubmit: (user, password) => this.authenticate(user, password)
        });
        return;
      }
    } catch (error: any) {
      this.views.authenticationView(
        "Unable to contact the server (" + error.message + ")",
        { onSubmit: (user, password) => this.authenticate(user, password) }
      );
      return;
    }
    // sesión inicializada: se muestra la autenticación
    this.showAuthentication();
  }

  // --------------------------------------------------------------------
  // vista de [autenticación]
  // --------------------------------------------------------------------
  private showAuthentication(error?: string): void {
    this.views.authenticationView(error, {
      onSubmit: (user, password) => this.authenticate(user, password)
    });
  }

  private async authenticate(user: string, password: string): Promise<void> {
    try {
      const response = await this.dao.authenticateUser(user, password);
      if (response.status === 200) {
        // autenticación exitosa: como en [main3.ts], ahora se recuperan,
        // de una vez por todas, los datos de impuestos
        // que necesita la capa [business]
        await this.initBusiness();
      } else {
        // fallo (login/contraseña incorrectos, o parámetros faltantes)
        this.showAuthentication(this.formatError(response.response));
      }
    } catch (error: any) {
      this.showAuthentication("Unable to contact the server (" + error.message + ")");
    }
  }

  // recupera los datos de la administración tributaria e instancia la
  // capa [business] con estos datos - una sola llamada al servidor para toda la sesión,
  // exactamente como [main3.ts]
  private async initBusiness(): Promise<void> {
    try {
      const response = await this.dao.getAdminData();
      if (response.status === 1000) {
        this.business = new Business(response.response);
        // nueva sesión de trabajo: se vuelve a empezar desde una lista vacía
        this.simulations = [];
        this.showCalculation();
      } else {
        this.showAuthentication(this.formatError(response.response));
      }
    } catch (error: any) {
      this.showAuthentication("Unable to contact the server (" + error.message + ")");
    }
  }

  // --------------------------------------------------------------------
  // vista de [cálculo de impuestos]
  // --------------------------------------------------------------------
  private showCalculation(result?: Simulation, error?: string): void {
    this.views.calculationView(result, error, {
      onSubmit: (married, children, salary) => this.calculate(married, children, salary),
      onList: () => this.showList(),
      onEndSession: () => this.endSession()
    });
  }

  // cálculo del impuesto realizado íntegramente en el navegador (capa [business]):
  // sin llamada al servidor, a diferencia de la primera versión de este cliente
  private calculate(married: string, children: number, salary: number): void {
    // this.business ha sido necesariamente inicializado: solo se llega a la
    // vista de [cálculo] después de [initBusiness]
    const result = this.business!.calculateTax(married, children, salary);
    // el resultado devuelto por [business.calculateTax] solo contiene los valores
    // calculados (tax, surcharge, discount, reduction, rate): se le agregan los parámetros
    // ingresados por el usuario, para que puedan
    // mostrarse de nuevo en la lista de simulaciones
    const simulation: Simulation = { married: married, children: children, salary: salary, ...result };
    // almacenamiento en el navegador (como `simulations.push(...)` en [main3.ts])
    this.simulations.push(simulation);
    this.showCalculation(simulation);
  }

  // --------------------------------------------------------------------
  // vista de [lista de simulaciones]
  // --------------------------------------------------------------------
  // ya no hay llamadas al servidor aquí: la lista ya está íntegramente en
  // [this.simulations]
  private showList(error?: string): void {
    this.views.listView(this.simulations, error, {
      onDelete: (number) => this.deleteSimulation(number),
      onCalculation: () => this.showCalculation(),
      onEndSession: () => this.endSession()
    });
  }

  private deleteSimulation(number: number): void {
    // eliminación local del arreglo [this.simulations], como
    // `simulations.splice(1, 1)` en [main3.ts] - tampoco hay llamada al servidor
    this.simulations.splice(number, 1);
    this.showList();
  }

  // --------------------------------------------------------------------
  // fin de sesión
  // --------------------------------------------------------------------
  private endSession(): void {
    // fin de sesión solo en el lado del cliente: se olvida la capa [business]
    // (y por tanto los datos de impuestos) y las simulaciones ya calculadas, y luego se
    // vuelve a la pantalla de autenticación. Este cliente no necesita
    // llamar al servidor para esto.
    this.business = undefined;
    this.simulations = [];
    this.showAuthentication();
  }

  // --------------------------------------------------------------------
  // da formato al campo [response] de una respuesta de error del servidor,
  // que es una cadena o un arreglo de cadenas (ver los controladores PHP)
  // --------------------------------------------------------------------
  private formatError(response: unknown): string {
    if (Array.isArray(response)) {
      return response.join(' ; ');
    }
    return String(response);
  }
}

// ejecución
window.addEventListener('DOMContentLoaded', () => {
  new AppHtml().start();
});

    • Línea 50: lo primero que hace la aplicación es llamar a init-session, que es obligatorio antes de cualquier otra acción (véase el archivo PHP [main.php], que bloquea cualquier acción hasta que se haya inicializado una sesión);

    • initBusiness (líneas 97–111), un método nuevo: llama a dao.getAdminData(); es el único lugar de todo el archivo donde se solicita esta acción al servidor, una vez por todas; si el estado es 1000, crea el objeto Business con los datos recibidos (línea 101, new Business(response.response)), reinicia el arreglo de simulaciones (línea 103; una nueva autenticación corresponde a una nueva sesión de trabajo) y, por último, muestra la vista de cálculo (línea 104).

    La vista de cálculo del impuesto (líneas 116–138):

    • showCalculation (líneas 116–122): sin cambios; se puede llamar con o sin un resultado: sin resultado para un formulario vacío (primera visita o regreso desde la lista), con resultado inmediatamente después de un cálculo exitoso;
    • calculate (líneas 126–138): es el método que más ha cambiado. Ya no es asíncrono y ya no contiene un bloque de red try/catch, puesto que ya no realiza ninguna llamada al servidor; en la línea 129, this.business!.calculateTax(married, children, salary) calcula directamente el resultado; el signo de exclamación (!) le indica a TypeScript que estamos seguros de que business ya fue inicializado (solo se puede llegar a esta vista después de initBusiness); en las líneas 130 a 134, como este resultado no contiene los parámetros ingresados, se le agregan mediante el operador de propagación ...result para construir el objeto simulation completo; en la línea 136, this.simulations.push(simulation) almacena esta nueva simulación en el arreglo del navegador; en la línea 137, la vista se vuelve a dibujar finalmente con este resultado.

    • showList (líneas 145–151): ya no realiza llamadas al servidor; pasa directamente this.simulations, el arreglo que ya está en memoria, a la capa [views];
    • deleteSimulation (líneas 153–158): en la línea 156, this.simulations.splice(number, 1) extrae la simulación directamente del arreglo del navegador, el equivalente exacto de la instrucción simulations.splice(1, 1) en [main3.ts]; luego, en la línea 157, se vuelve a mostrar la lista (actualizada).

    • endSession (líneas 163–171): ya no llama a dao.endSession() (este método se eliminó de [Dao3.ts]); simplemente descarta la capa [business] (línea 168, this.business = undefined) y las simulaciones (línea 169, this.simulations = []), y luego vuelve a mostrar la pantalla de autenticación (línea 170); como el resto de este cliente, finalizar la sesión ya no requiere, por lo tanto, ninguna solicitud de red;
    • formatError (líneas 177–182): sin cambios; el campo response de un error del servidor (PHP) es a veces una cadena y a veces un arreglo de cadenas (según el controlador); este método siempre reduce el mensaje a una sola cadena, lista para mostrarse;

    6.6. Conclusión

    Este capítulo ha mostrado cómo una misma base de lógica de negocio (las capas DAO y business ) puede ejecutarse tanto desde la línea de comandos (node.js, capítulos “Client HTTP 1” y “Client HTTP 2”) como en un navegador (capítulo “Client HTTP 3”), mediante una herramienta de compilación adecuada; en este caso, webpack y ts-loader. Esta base TypeScript es el punto de partida de los frameworks front-end modernos (React, Vue, Angular...), que se construyen sobre los mismos bloques: módulos, clases, async/await y una herramienta de compilación que los ensambla para el navegador.
