// Primer paso: conectarse a la base de datos MySQL [dbpersons] y luego
// desconectarse. Porta el script Python [mysql_01] del curso Python/Flask.
import mysql from 'mysql2/promise';

// conexión a la base de datos MySQL [dbpersons]
// la identidad del usuario es (admpersons, nobody)
const USER = 'admpersons';
const PWD = 'nobody';
const HOST = 'localhost';
const DATABASE = 'dbpersons';

// allá vamos
let connection: mysql.Connection | undefined;
try {
  console.log('Connecting to the MySQL DBMS...');
  // conexión
  connection = await mysql.createConnection({ host: HOST, user: USER, password: PWD, database: DATABASE });
  // continuación
  console.log(
    `Successful MySQL connection to database=${DATABASE}, host=${HOST} as user=${USER}, passwd=${PWD}`,
  );
} catch (error) {
  // mostrar el error
  console.log(`The following error occurred: ${error}`);
} finally {
  // cerrar la conexión si se abrió
  if (connection) {
    await connection.end();
  }
}

    Comentemos este código:

    • línea 16: [mysql.createConnection({ host: HOST, user: USER, password: PWD, database: DATABASE })] : abre una única conexión asíncrona; createConnection devuelve una Promise<Connection>, de ahí el await; es el equivalente en TypeScript de mysql.connector.connect(...) en el lado de Python;
    • línea 29: [connection.end()] : cierra la conexión de forma ordenada en el bloque finally, ocurra o no un error; la conexión se declara fuera del bloque try (tipo mysql.Connection | undefined) para seguir siendo accesible en este nivel.

    // databases/mysql/mysql-02.ts
// En este nuevo script, la conexión a la base de datos se aísla en una función.
// Porta el script Python [mysql_02].
import mysql from 'mysql2/promise';

// se conecta y luego se desconecta (login, pwd) de la base de datos [database] en el servidor [host]
// lanza una excepción si hay un problema
async function connect(host: string, database: string, login: string, pwd: string): Promise<void> {
  let cnx: mysql.Connection | undefined;
  try {
    // conexión
    cnx = await mysql.createConnection({ host, user: login, password: pwd, database });
    console.log(
      `Successful connection to database=${database}, host=${host} as user=${login}, passwd=${pwd}`,
    );
  } finally {
    // cerrar la conexión si se abrió
    if (cnx) {
      await cnx.end();
      console.log('Successful disconnection\n');
    }
  }
}

// - - - main
// credenciales de conexión
const USER = 'admpersons';
const PASSWD = 'nobody';
const HOST = 'localhost';
const DATABASE = 'dbpersons';

// conexión de un usuario existente
try {
  await connect(HOST, DATABASE, USER, PASSWD);
} catch (error) {
  // mostrar el error
  console.log(String(error));
}

// conexión de un usuario inexistente
try {
  await connect(HOST, DATABASE, 'xx', 'xx');
} catch (error) {
  // mostrar el error
  console.log(String(error));
}

    Comentemos este código:

    • línea 9: [async function connect(host, database, login, pwd): Promise<void>] : una función async que no devuelve nada pero puede lanzar una excepción; no captura por sí misma los errores de conexión, sino que los deja propagarse hacia quien la llama;
    • línea 40: [await connect(HOST, DATABASE, 'xx', 'xx')] : la segunda llamada, con un usuario “xx” que no existe en el lado de MySQL, lanza una excepción que el bloque catch envolvente muestra.

La causa no tiene relación con el script: para evitar revelar si una cuenta existe o no (protección contra la enumeración), MySQL simula un intercambio de autenticación al iniciar sesión con un usuario desconocido, y esta simulación está codificada para usar el plugin heredado mysql_native_password, que está deshabilitado por defecto desde MySQL 8.4. El principio que demuestra este script (una excepción lanzada e interceptada ante credenciales no válidas) permanece intacto; para recuperar el mensaje exacto mostrado, agregue `mysql_native_password=ON` debajo de `[mysqld]` en el archivo `my.ini` del servidor y luego reinicie el servidor.

    // databases/mysql/mysql-module.ts
// Funciones reutilizables para ejecutar una lista (o un archivo) de comandos SQL
// sobre una conexión MySQL abierta. Porta el módulo Python [mysql_module].
import type { Connection, FieldPacket, ResultSetHeader, RowDataPacket } from 'mysql2/promise';
import { readFile } from 'node:fs/promises';

// muestra el resultado de un comando SQL (resultado de un query() de mysql2)
// - si hay un arreglo [fields], era un SELECT: mostramos las columnas
//   y luego las filas;
// - en caso contrario, era una sentencia de actualización (INSERT/UPDATE/DELETE/DDL):
//   mostramos el número de filas modificadas (equivalente de cursor.rowcount).
function displayInfo(rows: RowDataPacket[] | ResultSetHeader, fields: FieldPacket[] | undefined): void {
  if (fields && fields.length > 0) {
    // era un SELECT - fields describe las columnas solicitadas
    const titles = fields.map((f) => f.name);
    console.log(titles.join(', '));
    console.log('*'.repeat(titles.join(', ').length));
    for (const row of rows as RowDataPacket[]) {
      console.log(titles.map((t) => row[t]));
    }
    console.log('*'.repeat(titles.join(', ').length));
  } else {
    // sin columnas descritas - sentencia de actualización
    const header = rows as ResultSetHeader;
    console.log(`number of modified rows: ${header.affectedRows}`);
  }
}

// usa la conexión abierta [connection]
// ejecuta sobre esta conexión los comandos SQL contenidos en la lista [sqlCommands]
// (un comando por elemento; las líneas vacías o que empiezan por # se ignoran)
// - si verbose=true, cada ejecución de una sentencia SQL va seguida de una visualización
//   que indica su éxito o su fallo;
// - si stop=true, la función se detiene en el 1.er error encontrado; en caso contrario
//   ejecuta todos los comandos;
// - si withTransaction=true, cualquier error cancela todas las sentencias SQL
//   ejecutadas antes (rollback); en caso contrario cada comando se confirma
//   de forma independiente (comportamiento por defecto de mysql2);
// la función devuelve un arreglo [error1, error2, ...]
export async function executeListOfCommands(
  connection: Connection,
  sqlCommands: string[],
  verbose = false,
  stop = true,
  withTransaction = true,
): Promise<string[]> {
  const errors: string[] = [];
  if (withTransaction) {
    await connection.beginTransaction();
  }
  try {
    for (const rawCommand of sqlCommands) {
      // elimina los espacios en blanco iniciales y finales del comando actual
      const command = rawCommand.trim();
      // ¿comando vacío o comentario? pasamos al siguiente
      if (command === '' || command[0] === '#') {
        continue;
      }
      // ejecución del comando actual
      try {
        const [rows, fields] = await connection.query(command);
        // sin error
        if (verbose) {
          console.log(`[${command}]: Successful execution`);
        }
        displayInfo(rows as RowDataPacket[], fields);
      } catch (error) {
        const msg = `${command}: Error (${error})`;
        errors.push(msg);
        if (verbose) {
          console.log(msg);
        }
        // ¿nos detenemos?
        if (withTransaction || stop) {
          return errors;
        }
      }
    }
    return errors;
  } finally {
    // confirmar / revertir la transacción si existe
    if (withTransaction) {
      if (errors.length > 0) {
        await connection.rollback();
      } else {
        await connection.commit();
      }
    }
  }
}

// usa la conexión abierta [connection]
// ejecuta sobre esta conexión los comandos SQL contenidos en el
// archivo de texto [sqlFilename] (un comando por línea)
export async function executeFileOfCommands(
  connection: Connection,
  sqlFilename: string,
  verbose = false,
  stop = true,
  withTransaction = true,
): Promise<string[]> {
  try {
    const content = await readFile(sqlFilename, 'utf-8');
    return await executeListOfCommands(connection, content.split('\n'), verbose, stop, withTransaction);
  } catch (error) {
    return [`The file ${sqlFilename} could not be processed: ${error}`];
  }
}

    Comentemos este código:

    • línea 45: [if (withTransaction) { await connection.beginTransaction(); }] : mysql2 está en modo AUTOCOMMIT de forma predeterminada (cada consulta se confirma individualmente); beginTransaction() cambia explícitamente la conexión al modo transaccional, tal como lo haría connexion.start_transaction() en el lado de Python;
    • línea 76: [if (withTransaction || stop) { return errors; }] : en modo transaccional, el primer error termina de inmediato el bucle; continuar sería inútil, ya que toda la transacción se revertirá de todos modos;
    • línea 82: [if (errors.length > 0) { await connection.rollback(); } else { await connection.commit(); }] : commit o rollback global, según haya ocurrido o no un error; véase, sin embargo, la trampa señalada en el script [mysql-04] más adelante.

    Precaución: Cabría esperar que la tabla `persons` esté vacía después de este `rollback()`; eso es lo que afirma, erróneamente, el curso original de Python sobre SQLAlchemy. En realidad, con MySQL/MariaDB (como con la mayoría de los SGBD), las instrucciones de definición de datos (DDL: CREATE TABLE, DROP TABLE...) confirman implícitamente (commit) y cierran cualquier transacción en curso antes de ejecutarse. Aquí, el CREATE TABLE del segundo comando ya ha confirmado, por lo tanto, (de manera invisible) todo lo anterior, incluido, puesto que no había nada antes, a sí mismo; los dos comandos INSERT que siguen abren entonces, en efecto, una nueva transacción implícita, que se revertirá efectivamente... salvo que, en la práctica, la verificación de la base de datos muestra que las dos filas insertadas siguen presentes después del rollback(): el CREATE TABLE confirmó la transacción mientras el bucle aún estaba en ejecución, incluso antes de que se ejecutaran las instrucciones INSERT posteriores; estas también quedan, por lo tanto, confirmadas de forma permanente en cuanto se ejecutan, fuera de cualquier transacción explícita. Recuerde la regla general: un DDL no puede revertirse con un ROLLBACK, y confirma silenciosamente cualquier transacción que estuviera abierta antes de él.

    // databases/mysql/mysql-05.ts
// Consultas parametrizadas: inserción preparada individual y luego inserción masiva.
// Porta el script Python [mysql_05]. Aquí volvemos a crear la tabla [persons] con
// el esquema que tiene una columna [id] (como en el script mysql-03), para que este script
// sea independiente del estado que deje mysql-04.
import mysql from 'mysql2/promise';

// la identidad del usuario
const ID = 'admpersons';
const PWD = 'nobody';
// la máquina anfitriona del DBMS
const HOST = 'localhost';
// identidad de la base de datos
const DATABASE = 'dbpersons';

type Person = [id: number, lastName: string, firstName: string, age: number];

// lista de personas (id, lastName, firstName, age)
const persons: Person[] = [];
for (let i = 0; i < 5; i++) {
  persons.push([i, `l0${i}`, `f0${i}`, i + 10]);
}
persons.push([40, "d'Aboot", "Y'éna", 18]);

// otra lista de personas
const otherPersons: Person[] = [];
for (let i = 0; i < 5; i++) {
  otherPersons.push([i + 100, `l1${i}`, `f1${i}`, i + 20]);
}
otherPersons.push([200, "d'Aboot", "F'ilhem", 34]);

// acceso al DBMS
let connection: mysql.Connection | undefined;
try {
  // conexión
  connection = await mysql.createConnection({ host: HOST, user: ID, password: PWD, database: DATABASE });

  // (re)creación de la tabla con el esquema de columna [id] (vea mysql-03.ts)
  await connection.query('drop table if exists persons');
  await connection.query(
    'create table persons (id int PRIMARY KEY, first_name varchar(30) NOT NULL, last_name varchar(30) NOT NULL, ' +
      'age integer NOT NULL, unique(last_name,first_name))',
  );

  // eliminación de los registros existentes
  await connection.query('delete from persons');
  // inserciones persona por persona con una sentencia preparada
  for (const person of persons) {
    const [id, lastName, firstName, age] = person;
    await connection.execute('insert into persons(id,last_name,first_name,age) values(?,?,?,?)', [id, lastName, firstName, age]);
  }
  // inserción masiva de una lista de personas - equivalente de executemany:
  // mysql2 ofrece connection.query('insert ... values ?', [[...],[...],...])
  // que construye una única consulta de varias filas
  const values = otherPersons.map(([id, lastName, firstName, age]) => [id, lastName, firstName, age]);
  await connection.query('insert into persons(id,last_name,first_name,age) values ?', [values]);

  // confirmación de la transacción - aquí cada consulta ya se confirmó automáticamente
  // (mysql2 está en modo AUTOCOMMIT por defecto, como el modo por defecto mencionado en
  // el script mysql-03; por lo tanto no hay nada que confirmar explícitamente)
  console.log(`${persons.length + otherPersons.length} persons inserted.`);
} catch (error) {
  console.log(`The following error occurred: ${error}`);
} finally {
  if (connection) {
    await connection.end();
  }
}

    7.2. Uso del ORM TypeORM

    Abordaremos ahora los ORM, inspirándonos en el capítulo “Uso del ORM SQLAlchemy” (capítulo 19) del curso de Python/Flask. El principio de los ORM (mapeo objeto-relacional) es manipular objetos del lenguaje en lugar de escribir SQL: una clase corresponde a una tabla, una instancia a una fila, un atributo a una columna y una relación entre clases (asociación, composición) a una clave externa.

    // databases/typeorm/typeorm-03-joined-queries.ts
// Muestra las consultas que TypeORM emite para obtener
// información repartida en varias tablas cuando se solicita una relación
// (@ManyToOne / @OneToMany). Retoma el script Python
// [main_joined_queries] (capítulo 19, scripts 05).
//
// A diferencia de sqlalchemy (usado por el curso de Python), TypeORM
// NO hace carga diferida por defecto en las entidades "clásicas" (repositorio):
// las relaciones por cargar deben solicitarse explícitamente, con la opción
// `relations`, o una consulta construida con `createQueryBuilder` /
// `leftJoinAndSelect`. Esta elección es deliberada y explícita: a diferencia de la
// carga diferida "oculta" de sqlalchemy, aquí siempre vemos, al leer el código,
// qué consultas SQL adicionales (joins) desencadenará una lectura.
import { AppDataSource } from './data-source-school.js';
import { SchoolClass } from './entities/school-class.entity.js';
import { Student } from './entities/student.entity.js';
import { Subject } from './entities/subject.entity.js';

await AppDataSource.initialize();

try {
  const studentRepo = AppDataSource.getRepository(Student);
  const classRepo = AppDataSource.getRepository(SchoolClass);
  const subjectRepo = AppDataSource.getRepository(Subject);

  // estudiante por id, con su clase (join explícito mediante `relations`)
  console.log('student id=11 -----------');
  const student11 = await studentRepo.findOneOrFail({ where: { id: 11 }, relations: { schoolClass: true } });
  console.log(`student={"id":${student11.id},"lastName":"${student11.lastName}","firstName":"${student11.firstName}"}`);
  console.log(`student's class: {"id":${student11.schoolClass.id},"name":"${student11.schoolClass.name}"}`);

  // los estudiantes de la misma clase (nueva consulta, con su propio join)
  console.log('students in the same class:');
  const studentsClass = await classRepo.findOneOrFail({
    where: { id: student11.schoolClass.id },
    relations: { students: true },
  });
  for (const s of studentsClass.students) {
    console.log(`student={"id":${s.id},"lastName":"${s.lastName}","firstName":"${s.firstName}"}`);
  }

  // un estudiante por su apellido
  console.log("student lastName='lastname2' -----------");
  const byName = await studentRepo.findOneOrFail({ where: { lastName: 'lastname2' }, relations: { schoolClass: true } });
  console.log(`student={"id":${byName.id},"lastName":"${byName.lastName}","firstName":"${byName.firstName}"}`);
  console.log(`student's class: {"id":${byName.schoolClass.id},"name":"${byName.schoolClass.name}"}`);

  // las notas de un estudiante, con la materia de cada nota
  console.log('grades of student id=11 -----------');
  const studentWithGrades = await studentRepo.findOneOrFail({
    where: { id: 11 },
    relations: { grades: { subject: true } },
  });
  for (const grade of studentWithGrades.grades) {
    console.log(`grade={"id":${grade.id},"value":${grade.value}}, subject={"name":"${grade.subject.name}"}`);
  }

  // los estudiantes de una clase, por nombre de clase
  for (const className of ['class1', 'class2']) {
    console.log(`students of class name='${className}' -----------`);
    const schoolClass = await classRepo.findOneOrFail({ where: { name: className }, relations: { students: true } });
    for (const s of schoolClass.students) {
      console.log(`{"id":${s.id},"lastName":"${s.lastName}","firstName":"${s.firstName}"}`);
    }
  }

  // las notas de una materia, por nombre de materia
  for (const subjectName of ['subject1', 'subject2']) {
    console.log(`subject with name='${subjectName}' -----------`);
    const subject = await subjectRepo.findOneOrFail({ where: { name: subjectName }, relations: { grades: true } });
    console.log(`subject={"name":"${subject.name}","coefficient":${subject.coefficient}}`);
    console.log('Grades in the subject: ');
    for (const grade of subject.grades) {
      console.log(`{"id":${grade.id},"value":${grade.value}}`);
    }
  }
} finally {
  await AppDataSource.destroy();
}

// Calcula, para un estudiante dado, su promedio ponderado por los coeficientes de
// las materias, así como su nota mínima y máxima. Retoma el script Python
// [main_stats_for_student] (capítulo 19, scripts 05).
//
// Trampa por conocer: las columnas `decimal` (aquí `grade.value` y
// `subject.coefficient`) se tipan como `number` en TypeScript dentro de
// la entidad, pero el controlador mysql2 en realidad las devuelve como
// *cadenas de caracteres* (para no perder precisión al
// convertirlas a un `number` de punto flotante). `typeof grade.value` es `'string'`
// en tiempo de ejecución, a pesar del tipo declarado. Un cálculo como
// `sum += grade.value * subject.coefficient` realizaría entonces una
// concatenación o una conversión implícita riesgosa: hay que convertir
// explícitamente con `Number(...)` antes de cualquier cálculo.
import { AppDataSource } from './data-source-school.js';
import { Student } from './entities/student.entity.js';

// calcula las estadísticas (promedio ponderado, mínimo, máximo) de un estudiante
async function studentStats(studentId: number): Promise<void> {
  const studentRepo = AppDataSource.getRepository(Student);
  // cargamos al estudiante con sus notas y, para cada nota, la materia
  // asociada (para tener el coeficiente) — una sola consulta con dos
  // joins, gracias a la opción `relations` anidada
  const student = await studentRepo.findOneOrFail({
    where: { id: studentId },
    relations: { grades: { subject: true } },
  });

  console.log(`Statistics of student {"id":${student.id},"lastName":"${student.lastName}","firstName":"${student.firstName}"}`);

  if (student.grades.length === 0) {
    console.log('  no grades.');
    return;
  }

  let weightedSum = 0;
  let coefficientsSum = 0;
  let min = Number.POSITIVE_INFINITY;
  let max = Number.NEGATIVE_INFINITY;

  for (const grade of student.grades) {
    // conversión explícita: `grade.value` y `grade.subject.coefficient`
    // son cadenas en tiempo de ejecución a pesar de su tipo declarado `number`
    const value = Number(grade.value);
    const coefficient = Number(grade.subject.coefficient);
    weightedSum += value * coefficient;
    coefficientsSum += coefficient;
    min = Math.min(min, value);
    max = Math.max(max, value);
    console.log(`  grade=${value.toFixed(2)} subject="${grade.subject.name}" (coefficient ${coefficient.toFixed(2)})`);
  }

  const average = weightedSum / coefficientsSum;
  console.log(`  weighted average=${average.toFixed(2)}, min=${min.toFixed(2)}, max=${max.toFixed(2)}`);
}

await AppDataSource.initialize();
try {
  // estudiante id=11: notas subject1=10 (coef 1.0), subject2=6 (coef 2.0)
  // promedio ponderado = (10*1.0 + 6*2.0) / (1.0+2.0) = 22/3 = 7.33
  await studentStats(11);
  console.log();
  // estudiante id=21: notas subject1=12 (coef 1.0), subject2=8 (coef 2.0)
  await studentStats(21);
  console.log();
  // estudiante id=32: notas subject1=14 (coef 1.0), subject2=10 (coef 2.0)
  await studentStats(32);
  console.log();
  // estudiante id=42: notas subject1=16 (coef 1.0), subject2=12 (coef 2.0)
  await studentStats(42);
} finally {
  await AppDataSource.destroy();
}

    7.2.7. Conclusión

    TypeORM permite escribir operaciones de lectura/escritura sin SQL, mediante entidades decoradas similares en espíritu a SQLAlchemy, al precio de una diferencia de diseño significativa que conviene conocer (no hay carga diferida implícita de relaciones: cualquier relación no solicitada explícitamente se omite) y de una trampa de herramientas propia de la ejecución con TSX (los tipos de columna deben especificarse explícitamente).     7.3. Uso del ORM Prisma

    Segundo ORM, con el mismo ejemplo del capítulo anterior (las bases de datos dbpersons y dbschool), para comparar los dos enfoques. Prisma se diferencia de TypeORM en su diseño: en lugar de anotar clases, el esquema de la base de datos se describe en un archivo aparte con el formato declarativo propio de Prisma, a partir del cual el comando prisma generate genera un cliente totalmente tipado.

    Importante: En el momento de redactar este capítulo, Prisma ORM está en la versión 7, que ha experimentado cambios internos significativos respecto de las versiones anteriores (5 y 6): el motor de consultas, antes un binario descargado por separado y compilado en Rust, es ahora un módulo WebAssembly integrado directamente en el paquete npm, combinado con un adaptador de driver (aquí @prisma/adapter-mariadb) que transmite las consultas a un driver JavaScript tradicional. Sin embargo, el principio general sigue siendo el mismo que en las versiones anteriores de Prisma, por lo que lo que sigue también se aplica a esas versiones.
    Nota sobre este entorno: el comando `npx prisma generate` , que genera el cliente TypeScript a partir del archivo schema.prisma, requiere un recurso de red específico de Prisma (binaries.prisma.sh) para obtener un componente interno de su CLI, un recurso al que la política de red del entorno de ejecución utilizado para preparar este curso no permitió el acceso. Por lo tanto, los scripts de este capítulo se escribieron y revisaron con el mismo cuidado que los anteriores, pero no pudieron ejecutarse ni verificarse en este entorno, a diferencia de los de los dos capítulos anteriores, todos los cuales se ejecutaron y cuyos resultados se verificaron. En una máquina con acceso normal a Internet (el caso típico), `npx prisma generate` se ejecuta sin ningún problema particular, y los scripts deberían producir las salidas “esperadas” que se muestran a continuación, calculadas a partir de los mismos datos ya verificados en el capítulo anterior.

// Calcula, para un estudiante dado, su promedio ponderado por los coeficientes de
// las materias, así como su nota mínima y máxima. Usa el mismo ejemplo que
// databases/typeorm/typeorm-04-stats.ts.
//
// Trampa por conocer, distinta de la de TypeORM: Prisma no devuelve
// las columnas `Decimal` (aquí `grade.value` y `subject.coefficient`) como
// cadenas, sino como instancias de la clase `Prisma.Decimal`
// (provista por la biblioteca decimal.js), para no perder ninguna
// precisión decimal. `typeof grade.value` es `'object'`, ni `'number'`
// ni `'string'`. Un cálculo como `grade.value * subject.coefficient`
// funcionaría casi por casualidad (`Decimal` redefine `valueOf`), pero
// la buena práctica — y la que se usa a continuación — es llamar
// explícitamente a `.toNumber()` antes de cualquier cálculo de punto flotante.
import { PrismaMariaDb } from '@prisma/adapter-mariadb';
import { PrismaClient } from '../../generated/prisma-school/client.js';

const adapter = new PrismaMariaDb({
  host: 'localhost',
  user: 'admschool',
  password: 'pwdschool',
  database: 'dbschool',
});
const prisma = new PrismaClient({ adapter });

// calcula las estadísticas (promedio ponderado, mínimo, máximo) de un estudiante
async function studentStats(studentId: number): Promise<void> {
  // una sola consulta, con dos niveles de join (notas, luego la materia de
  // cada nota) para obtener el coeficiente de cada materia
  const student = await prisma.student.findUniqueOrThrow({
    where: { id: studentId },
    include: { grades: { include: { subject: true } } },
  });

  console.log(`Statistics of student {"id":${student.id},"lastName":"${student.lastName}","firstName":"${student.firstName}"}`);

  if (student.grades.length === 0) {
    console.log('  no grades.');
    return;
  }

  let weightedSum = 0;
  let coefficientsSum = 0;
  let min = Number.POSITIVE_INFINITY;
  let max = Number.NEGATIVE_INFINITY;

  for (const grade of student.grades) {
    // conversión explícita: `grade.value` y `grade.subject.coefficient`
    // son valores `Prisma.Decimal`, no valores `number`
    const value = grade.value.toNumber();
    const coefficient = grade.subject.coefficient.toNumber();
    weightedSum += value * coefficient;
    coefficientsSum += coefficient;
    min = Math.min(min, value);
    max = Math.max(max, value);
    console.log(`  grade=${value.toFixed(2)} subject="${grade.subject.name}" (coefficient ${coefficient.toFixed(2)})`);
  }

  const average = weightedSum / coefficientsSum;
  console.log(`  weighted average=${average.toFixed(2)}, min=${min.toFixed(2)}, max=${max.toFixed(2)}`);
}

try {
  // estudiante id=11: notas subject1=10 (coef 1.0), subject2=6 (coef 2.0)
  // promedio ponderado = (10*1.0 + 6*2.0) / (1.0+2.0) = 22/3 = 7.33
  await studentStats(11);
  console.log();
  await studentStats(21);
  console.log();
  await studentStats(32);
  console.log();
  await studentStats(42);
} finally {
  await prisma.$disconnect();
}

    7.3.6. Conclusión

    Con el mismo ejemplo, TypeORM y Prisma producen código de longitud y legibilidad comparables; las diferencias son principalmente de estilo: clases decoradas y un Repository para TypeORM (más parecido a SQLAlchemy y a Spring Data JPA), y un esquema declarativo independiente y un cliente totalmente generado para Prisma (más parecido en espíritu a una herramienta como Alembic combinada con un generador de código). Ambos evitan escribir SQL y comparten la misma decisión de diseño, la ausencia de carga diferida implícita de relaciones, que los distingue de SQLAlchemy. La elección entre los dos, en un proyecto real, depende sobre todo del ecosistema (Prisma Studio, migraciones de Prisma, etc.) y de las preferencias del equipo.
